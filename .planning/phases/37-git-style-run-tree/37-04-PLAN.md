// server/autonomy/runs/runTreeBackfill.ts
import { storage } from '../../storage.js';
import { pool } from '../../db.js';   // raw pg pool — exception per eventLogger.ts:1-4

interface AutonomyEventRow {
  id: string;
  trace_id: string;
  project_id: string | null;
  user_id: string | null;
  hatch_id: string | null;
  event_type: string;
  timestamp: string;
  payload: Record<string, unknown> | null;
}

interface BackfillResult {
  runsCreated: number;
  stepsCreated: number;
  skipped: number;
}

/**
 * Backfill a single project's autonomy_events into autonomy_runs + autonomy_run_steps.
 * Idempotent: skips traceIds that already have at least one autonomy_run_steps row.
 * Applies 5-minute stable-event window per Pitfall 3 to avoid race with live writers.
 */
export async function backfillRunTreeForProject(projectId: string): Promise<BackfillResult> {
  let runsCreated = 0;
  let stepsCreated = 0;
  let skipped = 0;

  // Read events for this project, stable-window applied, ordered by trace_id + timestamp
  const eventsRes = await pool.query<AutonomyEventRow>(
    `SELECT id, trace_id, project_id, user_id, hatch_id, event_type, timestamp, payload
       FROM autonomy_events
      WHERE project_id = $1
        AND timestamp < NOW() - interval '5 minutes'
      ORDER BY trace_id, timestamp ASC`,
    [projectId],
  );

  // Group by trace_id
  const byTrace = new Map<string, AutonomyEventRow[]>();
  for (const ev of eventsRes.rows) {
    const list = byTrace.get(ev.trace_id) ?? [];
    list.push(ev);
    byTrace.set(ev.trace_id, list);
  }

  // For each (projectId, traceId) → check idempotency, then create run + steps
  for (const [traceId, events] of byTrace) {
    // Idempotency: skip if any autonomy_run_steps row already exists for this traceId
    const existing = await pool.query<{ count: string }>(
      `SELECT COUNT(*)::text as count FROM autonomy_run_steps WHERE trace_id = $1`,
      [traceId],
    );
    if (parseInt(existing.rows[0]?.count ?? '0', 10) > 0) {
      skipped += events.length;
      continue;
    }

    const firstEvent = events[0];
    const lastEvent = events[events.length - 1];
    const rootGoal = String((firstEvent.payload as any)?.taskTitle ?? (firstEvent.payload as any)?.taskId ?? 'Backfilled run');

    const run = await storage.createRun({
      traceId,
      projectId,
      userId: firstEvent.user_id,
      rootAgentId: firstEvent.hatch_id,
      rootGoal: rootGoal.slice(0, 500),
      status: 'complete',  // historical runs are complete; we have no in-flight signal from events alone
      stepCount: events.length,
      aggregateScoreDelta: null,
      metadata: {
        backfilledFrom: 'autonomy_events',
        flatHistorical: true,
        eventCount: events.length,
        firstTimestamp: firstEvent.timestamp,
        lastTimestamp: lastEvent.timestamp,
      },
    });
    runsCreated += 1;

    // Walk events in timestamp order; one step per event, all parentStepId=null (D-18 flat)
    for (const ev of events) {
      // D-19: score delta from payload-derived deliverableVersionId
      let scoreDelta: number | null = null;
      let deliverableVersionId: string | null = null;
      let deliverableVersionNumber: number | null = null;  // W-4 — denormalized so backfilled steps also open the correct version
      const dvId = (ev.payload as any)?.deliverableVersionId;
      if (typeof dvId === 'string') {
        deliverableVersionId = dvId;
        const scoreRes = await pool.query<{ rubric_score: any; deliverable_id: string; version_number: number }>(
          `SELECT rubric_score, deliverable_id, version_number FROM deliverable_versions WHERE id = $1 LIMIT 1`,
          [dvId],
        );
        const cur = scoreRes.rows[0];
        if (cur && Number.isFinite(cur.version_number)) {
          deliverableVersionNumber = cur.version_number;  // W-4 — captured from the same SELECT; no extra query
        }
        if (cur?.rubric_score?.total != null && Number.isFinite(cur.rubric_score.total)) {
          // Look up the prior version's rubric score
          const priorRes = await pool.query<{ rubric_score: any }>(
            `SELECT rubric_score FROM deliverable_versions
               WHERE deliverable_id = $1 AND version_number < $2
               ORDER BY version_number DESC LIMIT 1`,
            [cur.deliverable_id, cur.version_number],
          );
          const priorTotal = priorRes.rows[0]?.rubric_score?.total;
          if (Number.isFinite(priorTotal as number)) {
            scoreDelta = (cur.rubric_score.total as number) - (priorTotal as number);
          }
        }
      }
      // Derive step shape from event
      const stepType = inferStepTypeFromEvent(ev.event_type);
      const title = String((ev.payload as any)?.taskTitle ?? ev.event_type).slice(0, 200);
      await storage.createRunStep({
        runId: run.id,
        parentStepId: null,
        traceId,
        agentId: ev.hatch_id,
        agentName: String((ev.payload as any)?.agentName ?? '') || null,
        agentRole: String((ev.payload as any)?.agentRole ?? '') || null,
        stepType,
        title,
        status: 'complete',  // historical events are settled
        deliverableId: typeof (ev.payload as any)?.deliverableId === 'string' ? (ev.payload as any).deliverableId : null,
        deliverableVersionId,
        deliverableVersionNumber,  // W-4 — denormalized from the SELECT above; null when deliverableVersionId is null
        scoreDelta,
        metadata: { backfilledFromEventId: ev.id, eventType: ev.event_type } as Record<string, unknown>,
        completedAt: new Date(ev.timestamp),
        latencyMs: null,
      });
      stepsCreated += 1;
    }
  }

  return { runsCreated, stepsCreated, skipped };
}

function inferStepTypeFromEvent(eventType: string): 'task' | 'handoff' | 'peer_review' | 'deliberation' | 'safety_block' | 'approval_request' {
  if (eventType.startsWith('peer_review')) return 'peer_review';
  if (eventType.startsWith('handoff')) return 'handoff';
  if (eventType.includes('safety') || eventType.includes('block')) return 'safety_block';
  if (eventType.includes('approval')) return 'approval_request';
  if (eventType.includes('deliberation')) return 'deliberation';
  return 'task';
}

/**
 * Backfill all projects. Iterates projectIds via storage.getProjectsByUserId fan-out
 * (or via raw pool.query for safety — we don't need user attribution at this layer).
 */
export async function backfillAllProjects(): Promise<{ totalRuns: number; totalSteps: number; totalSkipped: number }> {
  let totalRuns = 0;
  let totalSteps = 0;
  let totalSkipped = 0;
  const res = await pool.query<{ id: string }>(`SELECT id FROM projects`);
  for (const { id: projectId } of res.rows) {
    const r = await backfillRunTreeForProject(projectId);
    totalRuns += r.runsCreated;
    totalSteps += r.stepsCreated;
    totalSkipped += r.skipped;
  }
  return { totalRuns, totalSteps, totalSkipped };
}

#!/usr/bin/env tsx
/**
 * Phase 37 — One-shot backfill of pre-Phase-37 autonomy_events into autonomy_runs + autonomy_run_steps.
 *
 * Expected runtime: up to ~5min for 100k events (most projects << that).
 * Idempotency: skips traceIds where any autonomy_run_steps row already exists.
 * 5-minute stable-event window: events newer than (NOW() - 5min) are SKIPPED to avoid
 *   race with the live writer (Pitfall 3 — backfill must NOT touch in-flight chains).
 *
 * Pre-Phase-37 history is reconstructed as FLAT lists per traceId. Handoff edges are
 * unrecoverable from the existing event catalog (D-18 — autonomy_events has no
 * `handoff_initiated` event type prior to 37-02). The reconstructed run rows are tagged
 * metadata.flatHistorical=true so the UI surfaces the "imported flat from history" hint.
 *
 * Run with: npm run backfill:run-tree
 *
 * D-20: MANUAL TRIGGER ONLY. Not auto-on-boot. Operator should run after deploying Phase 37
 * to backfill historical data, then never run again.
 */
import { backfillAllProjects } from '../server/autonomy/runs/runTreeBackfill.js';

async function main() {
  console.log('[backfill:run-tree] starting…');
  const startedAt = Date.now();
  const result = await backfillAllProjects();
  const elapsedMs = Date.now() - startedAt;
  console.log(`[backfill:run-tree] complete in ${elapsedMs}ms`);
  console.log(`  Runs created:  ${result.totalRuns}`);
  console.log(`  Steps created: ${result.totalSteps}`);
  console.log(`  Skipped (already-backfilled): ${result.totalSkipped}`);
}

main().catch((err) => {
  console.error('[backfill:run-tree] FAILED:', err);
  process.exit(1);
});
