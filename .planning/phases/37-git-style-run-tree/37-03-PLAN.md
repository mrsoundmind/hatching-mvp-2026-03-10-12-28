// shared/scoreFormat.ts
export type ScoreDeltaTone = 'positive' | 'negative' | 'neutral' | 'new' | 'silent';

export function formatScoreDelta(n: number | null | undefined): { label: string; tone: ScoreDeltaTone } {
  if (n === null || n === undefined) return { label: 'new', tone: 'new' };
  if (!Number.isFinite(n)) return { label: 'new', tone: 'new' };  // defensive — Pitfall 4
  if (n === 0) return { label: '', tone: 'silent' };
  if (n > 0) return { label: '+' + n.toFixed(1), tone: 'positive' };
  return { label: '−' + Math.abs(n).toFixed(1), tone: 'negative' };  // U+2212 en-dash, NOT ASCII hyphen
}
```

// client/src/hooks/useAutonomyRunTree.ts
import { useQuery, useQueryClient } from '@tanstack/react-query';
import type { AutonomyRun, AutonomyRunStep } from '@shared/schema';
import { useSidebarEvent, AUTONOMY_EVENTS } from './useSidebarEvent';  // verify exact import path

export function useAutonomyRunTree(projectId: string | undefined) {
  const queryClient = useQueryClient();

  const { data, isLoading } = useQuery<{ runs: AutonomyRun[]; steps: AutonomyRunStep[] }>({
    queryKey: ['/api/projects', projectId, 'runs'],
    enabled: !!projectId,
    staleTime: 30_000,
    refetchInterval: 30_000,
  });

  useSidebarEvent(AUTONOMY_EVENTS.TASK_COMPLETED, () => {
    if (projectId) queryClient.invalidateQueries({ queryKey: ['/api/projects', projectId, 'runs'] });
  });
  // If AUTONOMY_EVENTS.CHAIN_COMPLETED exists, do the same. Otherwise skip silently.

  return {
    runs: data?.runs ?? [],
    steps: data?.steps ?? [],
    isLoading,
  };
}

// client/src/components/sidebar/RunTreeNode.tsx
import { useMemo, useState } from 'react';
import { Clock, ArrowRightLeft, ShieldAlert, Search, MessagesSquare } from 'lucide-react';
import type { AutonomyRunStep } from '@shared/schema';
import { formatScoreDelta } from '@shared/scoreFormat';

const stepTypeIcons: Record<AutonomyRunStep['stepType'], typeof Clock> = {
  task: Clock,
  handoff: ArrowRightLeft,
  peer_review: Search,
  deliberation: MessagesSquare,
  safety_block: ShieldAlert,
  approval_request: ShieldAlert,
};

const stepTypeColor: Record<AutonomyRunStep['stepType'], string> = {
  task: 'var(--hatchin-blue)',
  handoff: 'var(--hatchin-text-muted)',
  peer_review: 'var(--hatchin-teal)',
  deliberation: 'var(--hatchin-text-muted)',
  safety_block: 'var(--hatchin-amber)',
  approval_request: 'var(--hatchin-amber)',
};

const MAX_VISIBLE_DEPTH = 3;

interface RunTreeNodeProps {
  step: AutonomyRunStep;
  allSteps: AutonomyRunStep[];
  depth: number;
  onClick: (step: AutonomyRunStep) => void;
}

export function RunTreeNode({ step, allSteps, depth, onClick }: RunTreeNodeProps) {
  const stepChildren = useMemo(
    () => allSteps.filter(s => s.parentStepId === step.id),
    [allSteps, step.id]
  );
  const [expanded, setExpanded] = useState(depth < MAX_VISIBLE_DEPTH);
  const StepIcon = stepTypeIcons[step.stepType] ?? Clock;
  const hasDeliverable = !!step.deliverableId;
  const delta = formatScoreDelta(step.scoreDelta);

  const handleClick = () => {
    if (hasDeliverable) {
      onClick(step);
    } else {
      setExpanded(!expanded);
    }
  };

  return (
    <div style={{ paddingLeft: `${Math.min(depth, MAX_VISIBLE_DEPTH) * 16}px` }}>
      <button
        onClick={handleClick}
        className="w-full flex items-center gap-2 px-2 py-1.5 rounded-lg hover:bg-[var(--hatchin-surface)] text-left transition-colors"
        aria-label={`Step ${step.title ?? step.stepType} — ${step.status}`}
      >
        <StepIcon className="w-3 h-3 shrink-0" style={{ color: stepTypeColor[step.stepType] }} />
        {/* 12px avatar with first letter of agentName */}
        <span
          className="w-3 h-3 rounded-full flex items-center justify-center text-[7px] font-bold shrink-0"
          style={{ backgroundColor: 'var(--hatchin-surface-elevated)', color: 'var(--hatchin-text)' }}
        >
          {step.agentName?.charAt(0) ?? '?'}
        </span>
        <span className="text-[11px] font-semibold shrink-0" style={{ color: 'var(--hatchin-text)' }}>{step.agentName ?? 'Unknown'}</span>
        <span className="text-[10px] hatchin-text-muted shrink-0">{step.agentRole ?? ''}</span>
        <span className="text-[11px] hatchin-text truncate flex-1">{step.title ?? '(no title)'}</span>
        {delta.tone === 'positive' && <span className="text-[10px] font-semibold" style={{ color: 'var(--hatchin-green)' }}>{delta.label}</span>}
        {delta.tone === 'negative' && <span className="text-[10px] font-semibold" style={{ color: 'var(--hatchin-amber)' }}>{delta.label}</span>}
        {delta.tone === 'new' && hasDeliverable && <span className="text-[9px] hatchin-text-muted">new</span>}
      </button>

      {!expanded && stepChildren.length > 0 && depth >= MAX_VISIBLE_DEPTH && (
        <button
          onClick={() => setExpanded(true)}
          className="ml-6 text-[10px] hatchin-text-muted"
        >
          ···{stepChildren.length} more
        </button>
      )}

      {expanded && stepChildren.map(child => (
        <RunTreeNode
          key={child.id}
          step={child}
          allSteps={allSteps}
          depth={depth + 1}
          onClick={onClick}
        />
      ))}
    </div>
  );
}

// client/src/components/sidebar/RunTreeView.tsx
import { ChevronDown, ChevronRight } from 'lucide-react';
import { useState } from 'react';
import { useAutonomyRunTree } from '../../hooks/useAutonomyRunTree';
import { RunTreeNode } from './RunTreeNode';
import { formatScoreDelta } from '@shared/scoreFormat';
import type { AutonomyRun, AutonomyRunStep } from '@shared/schema';

interface RunTreeViewProps {
  projectId: string | undefined;
}

export function RunTreeView({ projectId }: RunTreeViewProps) {
  const { runs, steps, isLoading } = useAutonomyRunTree(projectId);
  const [expandedRunIds, setExpandedRunIds] = useState<Set<string>>(() => new Set(runs.map(r => r.id).slice(0, 3)));

  const handleStepClick = (step: AutonomyRunStep) => {
    if (step.deliverableId) {
      window.dispatchEvent(new CustomEvent('open_deliverable', {
        detail: {
          deliverableId: step.deliverableId,
          // W-4: deliverableVersionNumber was denormalized onto the step row by the writer (37-02
          // completeStep) so we can dispatch the correct version directly — no second fetch.
          // null when this step did not produce a deliverable version (D-06.1 — common case for live).
          versionNumber: step.deliverableVersionNumber ?? undefined,
        },
      }));
    }
  };

  if (isLoading) return <div className="hatchin-text-muted text-[12px] p-3">Loading run tree…</div>;
  if (runs.length === 0) {
    return (
      <div className="px-3 py-8 text-center">
        <div className="text-[12px] hatchin-text-muted leading-snug">
          No autonomous runs yet.<br />
          Once a Hatch starts working on a task in the background, you'll see the run tree here.
        </div>
      </div>
    );
  }

  return (
    <div className="space-y-2 px-2">
      {runs.map(run => {
        const isOpen = expandedRunIds.has(run.id);
        const runSteps = steps.filter(s => s.runId === run.id);
        const rootSteps = runSteps.filter(s => s.parentStepId === null);
        const aggDelta = formatScoreDelta(run.aggregateScoreDelta);
        const isFlatHistorical = (run.metadata as Record<string, unknown>)?.flatHistorical === true;
        return (
          <div key={run.id} className="premium-card">
            <button
              onClick={() => setExpandedRunIds(prev => {
                const next = new Set(prev);
                if (next.has(run.id)) next.delete(run.id); else next.add(run.id);
                return next;
              })}
              className="w-full flex items-center gap-2 px-3 py-2"
              aria-expanded={isOpen}
            >
              {isOpen ? <ChevronDown className="w-3 h-3 shrink-0" /> : <ChevronRight className="w-3 h-3 shrink-0" />}
              <span className="text-[12px] truncate flex-1 text-left">{run.rootGoal ?? '(no goal recorded)'}</span>
              <span className="text-[10px] hatchin-text-muted shrink-0">{run.stepCount} steps</span>
              {aggDelta.tone === 'positive' && <span className="text-[10px] font-semibold" style={{ color: 'var(--hatchin-green)' }}>{aggDelta.label}</span>}
              {aggDelta.tone === 'negative' && <span className="text-[10px] font-semibold" style={{ color: 'var(--hatchin-amber)' }}>{aggDelta.label}</span>}
            </button>
            {isOpen && (
              <div className="pb-2">
                {rootSteps.map(s => (
                  <RunTreeNode key={s.id} step={s} allSteps={runSteps} depth={0} onClick={handleStepClick} />
                ))}
                {isFlatHistorical && (
                  <div className="px-3 py-1 text-[9px] hatchin-text-muted italic">
                    imported flat from history — newer runs will show full tree
                  </div>
                )}
              </div>
            )}
          </div>
        );
      })}
    </div>
  );
}
```
