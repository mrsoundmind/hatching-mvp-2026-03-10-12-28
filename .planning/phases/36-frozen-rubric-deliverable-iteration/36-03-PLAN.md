     * Phase 36 (RUBR-04) — Per-criterion rubric breakdown.
     *
     * Pure presentation. Consumes the rubricScore JSONB shape from a deliverable_versions row.
     *
     * IMPORTANT: This component MUST NOT import from @shared/deliverableRubrics — the rubric
     * registry's anchorAt0/anchorAt10 fields are server-only (T-36-02). We only need the
     * `criterion` key (snake_case), `score`, and `justification` that the server already
     * returns in rubricScore.breakdown.
     *
     * Three render states:
     *   1. null rubricScore  → "Score pending" (pre-Phase-36 version, or v1 baseline scoring failed)
     *   2. skipped: true     → "Rubric not available for this type" (legacy custom type)
     *   3. populated         → per-criterion list
     */
    import React from 'react';

    export interface RubricBreakdownProps {
      rubricScore: {
        total: number;
        breakdown: Array<{ criterion: string; score: number; justification: string }>;
        skipped?: boolean;
        reason?: string;
      } | null;
      rubricVersion: string | null;
    }

    /** Convert snake_case criterion key to Title Case for display. */
    function humanizeKey(key: string): string {
      return key
        .split('_')
        .map((w) => (w.length > 0 ? w[0].toUpperCase() + w.slice(1) : w))
        .join(' ');
    }

    export function RubricBreakdown({ rubricScore, rubricVersion }: RubricBreakdownProps) {
      if (!rubricScore) {
        return (
          <div className="p-3 rounded-lg bg-[var(--hatchin-surface)] text-xs hatchin-text-muted" data-testid="rubric-breakdown-empty">
            Score pending.
          </div>
        );
      }

      if (rubricScore.skipped) {
        return (
          <div className="p-3 rounded-lg bg-[var(--hatchin-surface)] text-xs hatchin-text-muted" data-testid="rubric-breakdown-skipped">
            Rubric not available for this type.
            {rubricScore.reason && <div className="mt-1 opacity-75">({rubricScore.reason})</div>}
          </div>
        );
      }

      return (
        <div className="p-3 rounded-lg bg-[var(--hatchin-surface)] space-y-3" data-testid="rubric-breakdown">
          <div className="flex items-baseline justify-between text-xs">
            <span className="font-medium hatchin-text">
              Score: {rubricScore.total.toFixed(1)} / 10
            </span>
            {rubricVersion && (
              <span className="hatchin-text-muted">rubric v{rubricVersion}</span>
            )}
          </div>
          <div className="space-y-2">
            {rubricScore.breakdown.map((row, i) => (
              <div key={`${row.criterion}-${i}`} className="text-xs" data-testid={`rubric-criterion-${row.criterion}`}>
                <div className="flex items-baseline justify-between gap-2">
                  <span className="hatchin-text font-medium">{humanizeKey(row.criterion)}</span>
                  <span className="hatchin-text-muted whitespace-nowrap">{row.score} / 10</span>
                </div>
                {row.justification && (
                  <div className="mt-0.5 hatchin-text-muted leading-relaxed">{row.justification}</div>
                )}
              </div>
            ))}
          </div>
        </div>
      );
    }
    ```

    - data-testid attributes: Playwright spec in 36-04 uses these for stable selectors. Three test IDs: `rubric-breakdown-empty`, `rubric-breakdown-skipped`, `rubric-breakdown` (populated state) + `rubric-criterion-{key}` for each row.
    - No icons, no emojis: project rule (CLAUDE.md § 14 — names matter for filenames, not for UI flair). The breakdown is information-dense and benefits from clean text, not iconography.
    - `toFixed(1)` on total: rubrics produce decimals; one decimal of precision matches the judge prompt's "rounded to 1 decimal" specification.
    - `key={row.criterion + '-' + i}` instead of just `key={i}`: deterministic across re-renders even if breakdown is reordered (defensive).

    import { motion } from 'framer-motion';
    import { X, ChevronDown, ChevronUp } from 'lucide-react';
    import { RubricBreakdown } from './RubricBreakdown';

    export interface AutoRevertBannerProps {
      oldScore: number;
      newScore: number;
      rejectedContent?: string;
      rejectedBreakdown?: Array<{ criterion: string; score: number; justification: string }>;
      rubricVersion?: string | null;
      onDismiss: () => void;
    }

    export function AutoRevertBanner({
      oldScore, newScore, rejectedContent, rejectedBreakdown, rubricVersion, onDismiss,
    }: AutoRevertBannerProps) {
      const [expanded, setExpanded] = useState(false);

      return (
        <motion.div
          initial={{ height: 0, opacity: 0 }}
          animate={{ height: 'auto', opacity: 1 }}
          exit={{ height: 0, opacity: 0 }}
          transition={{ type: 'spring', stiffness: 300, damping: 30 }}
          className="overflow-hidden"
          data-testid="auto-revert-banner"
        >
          <div className="mb-3 p-3 rounded-lg border bg-amber-50 dark:bg-amber-950/40 border-amber-300 text-amber-900 dark:text-amber-100">
            <div className="flex items-start gap-3">
              <div className="flex-1 min-w-0">
                <div className="text-sm font-medium">Refinement made it worse, kept previous version.</div>
                <button
                  type="button"
                  className="mt-1 text-xs underline-offset-2 hover:underline flex items-center gap-1"
                  onClick={() => setExpanded((e) => !e)}
                  data-testid="auto-revert-banner-expand"
                >
                  {expanded ? <ChevronUp className="w-3 h-3" /> : <ChevronDown className="w-3 h-3" />}
                  {expanded ? 'Hide details' : 'Click to see what changed.'}
                </button>
              </div>
              <button
                type="button"
                className="text-amber-900 dark:text-amber-100 hover:opacity-70 flex-shrink-0"
                onClick={onDismiss}
                aria-label="Dismiss banner"
                data-testid="auto-revert-banner-dismiss"
              >
                <X className="w-4 h-4" />
              </button>
            </div>
            {expanded && (
              <div className="mt-3 space-y-3" data-testid="auto-revert-banner-expanded-content">
                <div className="text-xs">
                  <span className="font-medium">Score:</span>{' '}
                  Old {oldScore.toFixed(1)} / 10 → New {newScore.toFixed(1)} / 10
                </div>
                {rejectedContent && (
                  <div className="text-xs">
                    <div className="font-medium mb-1">Rejected version:</div>
                    <pre className="whitespace-pre-wrap break-words p-2 rounded bg-amber-100/50 dark:bg-amber-900/30 max-h-40 overflow-y-auto leading-relaxed">{rejectedContent}</pre>
                  </div>
                )}
                <RubricBreakdown
                  rubricScore={{ total: newScore, breakdown: rejectedBreakdown ?? [] }}
                  rubricVersion={rubricVersion ?? null}
                />
              </div>
            )}
          </div>
        </motion.div>
      );
    }
    ```

    **Why each design choice:**
    - `motion.div` with `height: 0 → auto`: smooth slide-down on entry, slide-up on exit. Inline (not absolute-positioned) so it pushes content below it down rather than overlaying it — matches the "non-blocking" constraint.
    - `<pre className="whitespace-pre-wrap">` for rejected content: preserves markdown line breaks. max-h-40 + overflow-y-auto bounds the visual footprint so a long rejected document doesn't dominate the panel.
    - RubricBreakdown composed inside: reuses the Task 1 component. The breakdown shape `{total, breakdown}` is constructed from props because the rejected-version row may not have a full `skipped` field — just total + breakdown is sufficient.
    - Dismiss is `onDismiss` callback (not internal hide-self state): per D-15 dismissal is parent-controlled (revertBannerState lifted to ArtifactPanel) so a new revert event can re-mount the banner.
    - `aria-label="Dismiss banner"` on the X button: a11y requirement.
    - Tailwind dark-mode variants (`dark:bg-amber-950/40` etc.): matches existing project dark-mode support (verify by grepping ArtifactPanel for `dark:` patterns).
    - No focus-trap, no Esc-to-close: it's NOT a dialog. Keyboard users dismiss via Tab + Enter on the X button or simply ignore the banner.
    - data-testid plumbing: covers Playwright case 3 in 36-04 — banner-visible assertion + expand-toggle + dismiss assertion.
