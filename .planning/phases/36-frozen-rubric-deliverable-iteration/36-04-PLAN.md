    /**
     * Phase 36 (FBK-04) — Agent feedback signal aggregator.
     *
     * Builds a per-(projectId, agentId) snapshot of "your last N deliverables: X accepted,
     * Y dismissed, Z edited" for injection into the agent's system prompt. 60s in-process
     * cache per Q2 resolution — staleness is acceptable at the timescale of agent learning;
     * coupling-cost of write-invalidation exceeds the benefit.
     *
     * Threshold (D-23): below 3 finalized deliverables, returns null → the system prompt
     * omits the RECENT FEEDBACK section entirely (avoids noise from N=1 / N=2 signals).
     *
     * Privacy (D-24): aggregate counts ONLY — no IDs, no user names, no raw justifications.
     */
    import { storage } from '../storage.js';
    // (No import from ../../shared/deliverableTypes — labels are authored inline in TYPE_LABEL_MAP below for canonical-label correctness; see comment block. Per W-6.)

    export interface FeedbackSignal {
      total: number;
      accepted: number;
      dismissed: number;
      iterated: number;
      byType: Array<{
        type: string;
        accepted: number;
        dismissed: number;
        iterated: number;
        total: number;
      }>;
    }

    // === Type → Label map (built once at module load) ===
    // ============================================================================
    // Canonical user-facing label per deliverable type. SOURCE OF TRUTH.
    //
    // Why this is a literal map (NOT a for-loop over DELIVERABLE_TYPE_REGISTRY):
    //   Some slugs appear under MULTIPLE roles in DELIVERABLE_TYPE_REGISTRY with
    //   DIFFERENT labels (e.g. 'design-brief' is "Design Brief" under Product
    //   Designer, "UX Design Brief" under UX Designer; 'tech-spec' and 'gtm-plan'
    //   have similar duplicates). Building this map via .set() in iteration order
    //   produces last-wins behavior — deterministic but undocumented and brittle
    //   to registry reordering. This explicit map pins the canonical label per
    //   slug, independent of role ordering. Per W-6.
    //
    // 15 entries — one per unique non-`custom` type slug in shared/deliverableTypes.ts.
    // ('custom' is intentionally absent — agent-scoped feedback aggregation in
    // FBK-04 only operates over typed deliverables; 'custom' rows skip the section.)
    //
    // When adding a new deliverable type:
    //   1. Add the entry to shared/deliverableTypes.ts (one or more roles).
    //   2. Add ONE entry here with the canonical user-facing label.
    //   3. The acceptance-criterion `grep` gate below will require the map size
    //      to match the count of unique non-`custom` slugs.
    // ============================================================================
    const TYPE_LABEL_MAP: Readonly<Record<string, string>> = Object.freeze({
      'prd':                  'Product Requirements Document',
      'user-stories':         'User Stories',
      'project-plan':         'Project Plan',
      'tech-spec':            'Technical Specification',
      'design-brief':         'Design Brief',
      'gtm-plan':             'Go-to-Market Plan',
      'blog-post':            'Blog Post',
      'landing-copy':         'Landing Page Copy',
      'content-calendar':     'Content Calendar',
      'email-sequence':       'Email Sequence',
      'seo-brief':            'SEO Strategy Brief',
      'competitive-analysis': 'Competitive Analysis',
      'market-research':      'Market Research Report',
      'process-doc':          'Process Documentation',
      'data-report':          'Data Analysis Report',
    });

    function getTypeLabel(type: string): string {
      return TYPE_LABEL_MAP[type] ?? type;
    }

    // === Cache (module-scope, 60s TTL) ===
    const CACHE_TTL_MS = 60_000;
    const CACHE_MAX_ENTRIES = 1000;
    const cache = new Map<string, { value: FeedbackSignal | null; expiresAt: number }>();

    function pruneIfFull(now: number): void {
      if (cache.size <= CACHE_MAX_ENTRIES) return;
      for (const [k, v] of cache) {
        if (v.expiresAt < now) cache.delete(k);
      }
    }

    // === Public API ===
    export async function getRecentFeedbackSignal(
      projectId: string,
      agentId: string,
    ): Promise<FeedbackSignal | null> {
      const key = `${projectId}:${agentId}`;
      const now = Date.now();
      const hit = cache.get(key);
      if (hit && hit.expiresAt > now) {
        return hit.value;
      }

      const rows = await storage.getRecentFinalizedDeliverablesByAgent(projectId, agentId, 10);

      // D-23: threshold — below 3 finalized → null (cached for TTL so subsequent calls don't re-hit storage)
      if (rows.length < 3) {
        pruneIfFull(now);
        cache.set(key, { value: null, expiresAt: now + CACHE_TTL_MS });
        return null;
      }

      const byTypeMap = new Map<string, { accepted: number; dismissed: number; iterated: number; total: number }>();
      let accepted = 0, dismissed = 0, iterated = 0;
      for (const r of rows) {
        const stats = byTypeMap.get(r.type) ?? { accepted: 0, dismissed: 0, iterated: 0, total: 0 };
        stats.total += 1;
        if (r.userAcceptedAt) { stats.accepted += 1; accepted += 1; }
        if (r.dismissedAt)   { stats.dismissed += 1; dismissed += 1; }
        if ((r.editsCount ?? 0) > 0) { stats.iterated += 1; iterated += 1; }
        byTypeMap.set(r.type, stats);
      }

      const byType = [...byTypeMap.entries()]
        .map(([type, stats]) => ({ type, ...stats }))
        .sort((a, b) => b.total - a.total);   // most-common types first

      const signal: FeedbackSignal = {
        total: rows.length,
        accepted,
        dismissed,
        iterated,
        byType,
      };
      pruneIfFull(now);
      cache.set(key, { value: signal, expiresAt: now + CACHE_TTL_MS });
      return signal;
    }

        lines.push(`Your last ${t.total} ${labelPluralized}: ${t.accepted} accepted, ${t.dismissed} dismissed, ${t.iterated} edited.`);

      // Optional trend line if dismissal rate > 50%.

      if (dismissalRate > 0.5) {
        lines.push("Trend: more than half were dismissed. Tighten specificity and risk coverage; the user is signaling these aren't landing.");
      }
      lines.push('Let this inform what you produce next without quoting it.');
      return lines.join('\n');
    }

      if (process.env.NODE_ENV === 'production') {

      }
      cache.clear();
    }

    ## Known Limitations

    1. **No backfill of pre-Phase-36 deliverable_versions rows** (Q3 resolution): rows created before this phase have `rubricVersion = NULL` and `rubricScore = NULL`. The 36-03 RubricBreakdown component renders the empty state for these. No data loss; observable in the version history.

Client-side `disabled={iterateMutation.isPending}` is the user-facing fix. Transactional iterate is deferred.

    5. **Feedback signal cache is 60s in-process** (Q2 resolution): no write-invalidation on Accept/Dismiss. Worst-case staleness is 60s. Per the rationale in RESEARCH §Q2, the signal is meant to inform behavior over many turns; sub-minute precision is unnecessary.
